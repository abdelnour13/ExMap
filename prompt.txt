Goal : Build a backend service to be later used by a frontend application to interact with the possible expertise maps.

Requirements : 
    - Uses FastAPI
    - NEO4J credentials loaded from a .env file.
    - dockerized application (dockerfile and docker-compose) and a requirements.txt file.
    - Proper HTTP errors handling and responses.
    - For visualization queries use python package 'neo4j_vis' to generate and HTML file which will be stored
    under a proper sub-folder of /public its name generated by concatinating the timestamp and a uuid and the response should contain
    the link to the generated file. 
    - The following endpoints :
        - GET /graphs : returns a list of summaries of all the available expertise maps it is expected that only handfull
        of expertises will be available so no need for pagination. each summary contains : 'id', 'definition', 'number of expertises',
        'list of labs', 'list of sub-domains'.
        - GET /graphs/{id} : same informations as the /graphs but for the specific graph.
        - GET /graphs/{id}/download : downloads the graph as a json dump.
        - POST /graphs/{id}/vis/taxonomy : displays the domain-subdomain links.
        - POST /graphs/{id}/vis/expertises : displays the expertises<->subdomains and subdomains<->domain links.
        - POST /graphs/{id}/vis/expertises/{expertise-id} : diaplays the following links for the specific expertises :
            expertise<->entity
            expertise<->subdomain
            subdomain<->domain
            entity<->subdomain

Example of neo4j_vis usage:

def visualize(records: list[Record]) -> HTML:

    triple_ids = set()
    nodes = []
    relationships = []

    params = {
        "Domain" : {
            "size" : 20,
            "color" : "#F99CFF"
        },
        "Entity" : {
            "size" : 10,
            "color" : "#B97263"
        }
    }

    for record in records:

        data = record.data()

        triples : list[tuple[dict, dict, dict]] = [data['r']] if 'r' in data else [data['r1'], data['r2']]

        for src, rel, dst in triples:

            id_ = f"{src['id']}:{rel}:{dst['id']}"

            if id_ in triple_ids:
                continue
            else:
                triple_ids.add(id_)
            
            ### Remove Unwanted Arguments
            src_props = src.copy()
            dst_props = dst.copy()

            for node in [src_props, dst_props]:
                node.pop('NodeType', None)
                node.pop('level', None)
                node.pop('score', None)
                node.pop('NodeType', None)
                node.pop('id', None)

            ### Nodes
            src_node = Node(id=src["id"], caption=src["name"], **params[src["NodeType"]], properties=src_props)
            dst_node = Node(id=dst["id"], caption=dst["name"], **params[dst["NodeType"]], properties=dst_props)
            nodes.extend([src_node, dst_node])

            ### Relationships
            relation = Relationship(source=src["id"], target=dst["id"], caption=rel)
            relationships.append(relation)

    VG = VisualizationGraph(nodes=nodes, relationships=relationships)
    return VG.render()


def save_file(html: HTML, filename: str = "graph.html") -> None:
    with open(filename, "w", encoding="utf-8") as f:
        f.write(html.data)
